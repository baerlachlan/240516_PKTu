gene_gclen <- transcripts %>%
    mcols() %>%
    as_tibble() %>%
    group_by(gene_id) %>%
    summarise(
        gc_content = sum(gc_content*length) / sum(length),
        length = ceiling(median(length))
    )

vcf <- here("smk-variants/results/variation/vcf/snvs.final.vcf")
groups <- list(
    eofad = meta$sample[meta$geno_psen1 == "eofad"],
    fai = meta$sample[meta$geno_psen1 == "fai"],
    wt = meta$sample[meta$geno_psen1 == "wt"],
    tu = meta$sample[meta$geno_chr14 == "tu"],
    pk = meta$sample[meta$geno_chr14 == "pk"],
    het = meta$sample[meta$geno_chr14 == "het"]
)
contrasts <- makeContrasts(
    eofad_wt = eofad - wt,
    fai_wt = fai - wt,
    eofad_fai = eofad - fai,
    pk_tu = pk - tu,
    tu_het = tu - het,
    pk_het = pk - het,
    levels = names(groups)
)
Code 
gts <- tadar::readGenotypes(vcf, genome = genome(genes)) %>%
    subsetByOverlaps(as(seqinfo(genes), "GRanges")) %>%
    keepStandardChromosomes(pruning.mode = "coarse")
props <- gts %>%
    tadar::countAlleles(groups = groups) %>%
    tadar::filterLoci(n_called == 18) %>%
    tadar::countsToProps()
dar_11loci <- dar(props, contrasts = contrasts, region_loci = 5)
saveRDS(dar_11loci, here("Rdata/dar_11loci.Rds"))
