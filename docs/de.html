lb_complex_dt <- function(tbl, cap, rownames = FALSE){
    datatable(
        tbl,
        caption = htmltools::tags$caption(
            cap,
            style = "text-align: left;"
        ),
        options = list(
            dom = "tflipr",  # Available options: tflipr
            columnDefs = list(list(className = 'dt-left', targets = "_all")),  # Align all columns left
            pageLength = 10,  # Show all results
            scrollCollapse = TRUE,  # Removes whitespace when table is small
            scrollY = "100%",  # Sets max height of table before scrollbar is added
            scrollX = TRUE,  # Enable horizontal scrolling when too wide
            sScrollX = "100%",  # Stops last column title being cut off
            initComplete = JS(  # Set font
                "function(settings, json) {",
                "$('body').css({'font-family': 'Calibri'});",
                "}"
            )
        ),
        rownames = rownames,
        height = "100%",  # Seems to work in tandem with scrollCollapse
        width = "100%",  # Fills width when viewed in RStudio viewer
        class = "stripe hover row-border order-column compact nowrap"
    )
}

Code 
min_cpm <- 1
min_samps <- meta %>%
    group_by(geno_joint) %>%
    summarise(n = n()) %>%
    pull(n) %>%
    min()
dge_samples <- tibble(sample = colnames(counts)) %>%
    left_join(meta)
dge_genes <- genes %>%
    as.data.frame() %>%
    .[rownames(counts),] %>%
    dplyr::select(chr = seqnames, everything()) %>%
    mutate(chr = fct_relevel(chr, primary_chrs))
dge_unfilt <- DGEList(
    counts,
    samples = dge_samples,
    genes = dge_genes
) %>%
    calcNormFactors()
keep_genes <- rowSums(cpm(dge_unfilt$counts) >= min_cpm) >= min_samps
dge_list <- dge_unfilt[keep_genes, , keep.lib.sizes=FALSE] %>%
    calcNormFactors()

Gene-level count data was loaded from featureCounts and imported into R as a DGEList object.

Genes with low counts across all libraries provide little evidence for differential expression. The discreteness of these counts also interferes with downstream statistical approximations. Genes were therefore retained for downstream analysis if they had 1 CPM in at least 3 samples. Filtering is performed independently of which sample belongs to which group so that no bias is introduced.

18134 genes remained for further analysis with library sizes ranging from 60,946,600 to 160,656,710.

Model fitting

Code 
fit_psen1 <- estimateDisp(dge_list, design_psen1) %>%
    glmQLFit()
tt_psen1 <- colnames(design_psen1)[2:3] %>%
    sapply(function(x){
        glmQLFTest(fit_psen1, coef = x) %>%
            topTags(n = Inf) %>%
            .[["table"]] %>%
            as_tibble() %>%
            arrange(PValue) %>%
            dplyr::select(
                gene_id, gene_name, logFC, logCPM, PValue, FDR, everything()
            ) %>%
            mutate(
                bonfP = p.adjust(PValue, "bonf"),
                DE = ifelse(FDR < 0.05, TRUE, FALSE),
                DE_cat = case_when(
                    logFC <= 0 & DE ~ "Downregulated",
                    logFC >= 0 & DE ~ "Upregulated",
                    !DE ~ "Non-siginificant"
                ),
                coef = x,
                label = glue("<b style='color:{pal_group[coef]}'>{coef}</b>"),
            )
    }, simplify = FALSE)

fit_chr14 <- estimateDisp(dge_list, design_chr14) %>%
    glmQLFit()
tt_chr14 <- colnames(design_chr14)[2:3] %>%
    sapply(function(x){
        glmQLFTest(fit_chr14, coef = x) %>%
            topTags(n = Inf) %>%
            .[["table"]] %>%
            as_tibble() %>%
            arrange(PValue) %>%
            dplyr::select(
                gene_id, gene_name, logFC, logCPM, PValue, FDR, everything()
            ) %>%
            mutate(
                bonfP = p.adjust(PValue, "bonf"),
                DE = ifelse(FDR < 0.05, TRUE, FALSE),
                DE_cat = case_when(
                    logFC <= 0 & DE ~ "Downregulated",
                    logFC >= 0 & DE ~ "Upregulated",
                    !DE ~ "Non-siginificant"
                ),
                coef = x,
                label = glue("<b style='color:{pal_group[coef]}'>{coef}</b>"),
            )
    }, simplify = FALSE)
